from tensorflow import keras

# Load the saved model
model = keras.models.load_model("satellite_classifier.keras")

# Load class names (if you saved them)
import json
with open("class_names.json", "r") as f:
    class_names = json.load(f)
#################################################################################################

#In the 2 nd part run this code for testing images:

import gradio as gr
import tensorflow as tf
import numpy as np

def predict(image):
    img = tf.image.resize(image, (128, 128))
    img = tf.expand_dims(img, axis=0)
    pred = model.predict(img)
    class_idx = np.argmax(pred[0])
    class_name = class_names[class_idx]
    confidence = float(pred[0][class_idx])
    return {class_name: confidence}

interface = gr.Interface(
    fn=predict,
    inputs=gr.Image(type="numpy"),
    outputs=gr.Label(num_top_classes=3),
    title="Satellite Image Classifier",
    description="Upload a satellite image to classify it."
)

interface.launch()
